# shared cases for the editing toolbar's commands (docs/technology-decisions.md, t17). the
# swift core tests (ios/Tests/EditorCommandsTests.swift) and the kotlin core tests
# (android/core-tests/.../EditorCommandsTest.kt) both read this file.
#
# each case is one line: the command, the text before, and the text after, separated by spaces.
# the texts are in double quotes, with \n for a line feed, \r for a carriage return, \t for a
# tab, \" for a quote, and \\ for a backslash. ‸ is the caret, and « and » mark a selection. an
# after text of - means the command changes nothing. empty lines and lines that start with #
# are skipped.
#
# indent, outdent, bold, and italic come from laperm on ios. their cases describe laperm's
# behavior at the pinned commit (from its own tests) and run only in the kotlin tests. task,
# link, and tag run in both.

# indent and outdent: list lines only, four spaces.
indent   "- it‸em"                       "    - it‸em"
indent   "1. fi‸rst"                     "    1. fi‸rst"
indent   "2) se‸cond"                    "    2) se‸cond"
indent   "- [ ] t‸ask"                   "    - [ ] t‸ask"
indent   "«- a\n- b»"                    "    «- a\n    - b»"
indent   "«- a\nplain\n- b»"             "    «- a\nplain\n    - b»"
indent   "- a\r\n- b‸"                   "- a\r\n    - b‸"
indent   "pl‸ain"                        -
indent   "-no space‸"                    -
indent   "‸"                             -
outdent  "    - it‸em"                   "- it‸em"
outdent  "  - ‸a"                        "- ‸a"
outdent  "- ‸a"                          -
outdent  "«    - a»"                     "«- a»"
outdent  " «   - a\n    »- b\ntail"      "«- a\n»- b\ntail"
outdent  "        - de‸ep"               "    - de‸ep"
# a tab is not removed, and indent adds spaces before it (laperm's rule).
outdent  "\t- it‸em"                     -
indent   "\t- it‸em"                     "    \t- it‸em"
